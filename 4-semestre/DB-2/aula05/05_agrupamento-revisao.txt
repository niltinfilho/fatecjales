-- 37
select
	count(codigo_cliente) as quantidade, 
	uf 
from cliente
group by uf;

-- 38
select 
	count(codigo_vendedor) as quantidade_funcionario, 
	sum(salario_fixo) as soma_salario,
	faixa_comissao
from vendedor
group by faixa_comissao
order by faixa_comissao;

-- 39
select
	sum(val_unit) soma_valor, 
	unidade
from produto
group by unidade;

-- 40 
select
	count(num_pedido) as quantidade_pedido,
	codigo_vendedor
from pedido
where prazo_entrega > 7
group by codigo_vendedor, prazo_entrega;

-- 41
select count(codigo_cliente) as quantidade_clientes, uf
from cliente
where uf != 'SP'
group by uf;

-- 42
select 
	p.num_pedido,
	p.codigo_cliente,
	p.codigo_vendedor,
	ip.quantidade
from pedido p
inner join item_pedido ip
	on ip.num_pedido = p.num_pedido
where ip.quantidade > 3;

-- 43
select 
	p.codigo_produto,
	p.descricao_produto,
	sum(ip.quantidade) as soma_quantidade
from item_pedido ip
inner join produto p
	on ip.codigo_produto  = p.codigo_produto
group by p.codigo_produto, p.descricao_produto
having sum(ip.quantidade) >= 50;

-- 44
select
	p.unidade,
	avg(p.val_unit) as media_val_unit
from produto p
group by p.unidade
having avg(p.val_unit) < 1.5;

-- 45
select
	c.nome_cliente,
	c.cidade
from cliente c
inner join pedido p
	on p.codigo_cliente = c.codigo_cliente
where p.prazo_entrega > 25;

-- 46
select 
	p.codigo_produto,
	p.descricao_produto,
	p.unidade,
	sum(ip.quantidade) as quantidade
from item_pedido ip
inner join produto p
	on ip.codigo_produto = p.codigo_produto
group by p.codigo_produto 
having sum(ip.quantidade) > 40;

-- 47
select distinct c.nome_cliente
from pedido p
inner join cliente c
	on p.codigo_cliente = c.codigo_cliente
where p.codigo_vendedor = 101;

-- 48
select distinct v.nome_vendedor
from pedido p
inner join vendedor v
	on p.codigo_vendedor = v.codigo_vendedor
where p.codigo_cliente = 720;

-- 49
select 
	c.nome_cliente, 
	p.prazo_entrega
from pedido p
inner join cliente c
	on p.codigo_cliente = c.codigo_cliente
order by c.nome_cliente desc;

-- 50
select distinct
	v.nome_vendedor,
	p.prazo_entrega, 
	v.salario_fixo
from pedido p
inner join vendedor v
	on p.codigo_vendedor = v.codigo_vendedor
where p.prazo_entrega > 15 and v.salario_fixo >= 1000
order by v.nome_vendedor;

-- 51
select distinct
	c.nome_cliente,
	p.prazo_entrega,
	c.uf
from pedido p
inner join cliente c
	on p.codigo_cliente = c.codigo_cliente
where p.prazo_entrega > 15 and c.uf = 'SP' or c.uf = 'RJ';

-- 52
select c.nome_cliente
from pedido p
inner join cliente c
	on p.codigo_cliente = c.codigo_cliente
where p.codigo_vendedor = (select codigo_vendedor from vendedor where nome_vendedor = 'Antônio')
order by c.nome_cliente;

-- 53
select c.nome_cliente
from pedido p
inner join cliente c
	on c.codigo_cliente = p.codigo_cliente
inner join item_pedido ip
	on ip.num_pedido = p.num_pedido
inner join produto pr
	on pr.codigo_produto = ip.codigo_produto
where p.prazo_entrega > 15 and pr.descricao_produto = 'Queijo' and c.uf = 'RJ';

-- 54
select 
	c.codigo_cliente, 
	c.nome_cliente,
	p.prazo_entrega
from cliente c
left join pedido p
	on c.codigo_cliente = p.codigo_cliente;

-- 55
select
    v.codigo_vendedor,
    v.nome_vendedor,
    p.num_pedido
from vendedor v
left join pedido p
    on v.codigo_vendedor = p.codigo_vendedor;

-- 56
select
    p.codigo_produto,
    p.descricao_produto,
    ip.num_pedido,
    ip.quantidade
from produto p
left join item_pedido ip
    on p.codigo_produto = ip.codigo_produto;

-- 57
select nome_cliente
from cliente
where codigo_cliente in (
    select codigo_cliente
    from pedido
    where prazo_entrega > 25
);

-- 58
select
    codigo_cliente,
    nome_cliente
from cliente
where codigo_cliente not in(
    select codigo_cliente
    from pedido
);

-- 59
select descricao_produto
from produto
where val_unit < (
    select avg(val_unit)
    from produto
);

-- 60
select codigo_cliente
from pedido
where prazo_entrega > (
    select avg(prazo_entrega)
    from pedido
);

-- 61
select c.nome_cliente
from cliente c
inner join pedido p
    on c.codigo_cliente = p.codigo_cliente
where p.prazo_entrega > (
    select avg(prazo_entrega)
    from pedido
);

-- 62
select p.descricao_produto
from produto p
inner join item_pedido ip
    on p.codigo_produto = ip.codigo_produto
where ip.quantidade = (
    select max(quantidade)
    from item_pedido
);

-- 63
select distinct c.nome_cliente
from cliente c
inner join pedido p
    on c.codigo_cliente = p.codigo_cliente
inner join item_pedido ip
    on p.num_pedido = ip.num_pedido
where ip.quantidade > (
    select avg(quantidade)
    from item_pedido
);
